[6] GET /books
    └─ List all books with filtering & pagination
    └─ Query Parameters:
       • page: Page number (default: 1)
       • limit: Items per page (default: 10, max: 100)
       • author_id: Filter by author (optional)
       • title: Search by title (partial match, optional)
       • tags: Filter by tags (optional, can be multiple)
    └─ Usage Examples:
       ① All books: curl -s http://localhost:8000/books
       ② Page 2, 20 per page: curl -s "http://localhost:8000/books?page=2&limit=20"
       ③ By author: curl -s "http://localhost:8000/books?author_id=1"
       ④ Search title: curl -s "http://localhost:8000/books?title=Python"
       ⑤ By tags: curl -s "http://localhost:8000/books?tags=Fiction&tags=Classic"
       ⑥ Combined: curl -s "http://localhost:8000/books?author_id=1&title=Python&page=1&limit=10"

BASE_URL="http://localhost:8000"

# 5. Get Book

curl -s "$BASE_URL/books/1" | jq .

# 6. List Authors
echo -e "\n6️⃣  List Authors"
curl -s "$BASE_URL/authors" | jq .

# 7. Author's Books
echo -e "\n7️⃣  Get Author's Books"
curl -s "$BASE_URL/authors/1/books" | jq .

# 8. Create Book
echo -e "\n8️⃣  Create Book"
curl -X POST "$BASE_URL/books" \
  -H "Content-Type: application/json" \
  -d "{\"id\": $NEXT_ID, \"title\": \"Test\", \"author_id\": 1, \"publisher\": \"Test\", \"pages\": 300}" | jq .

echo -e "\n9️⃣  Publisher Stats"

from frontend.api_client import get_client

client = get_client()

# Get next ID
next_id = client.get_next_book_id()["next_id"]
print(f"Creating book with ID: {next_id}")

# Create book
book_data = {
    "id": next_id,
    "title": "The Great Gatsby",
    "author_id": 1,
    "publisher": "Penguin",
    "pages": 180,
    "tags": ["Fiction"]
}
response = client.create_book(book_data)
print(f"Book created: {response}")

# List books with filter
books = client.list_books(author_id=1, page=1, limit=10)
for book in books["items"]:
    print(f"- {book['title']} ({book['pages']} pages)")

# Update book
update_data = {"title": "Updated Title", "pages": 200}
client.update_book(book_id=1, book_data=update_data)

# Delete book
client.delete_book(book_id=1)

# Author operations
authors = client.list_authors()
books_by_author = client.get_author_books(author_id=1)

# Publisher stats
stats = client.get_publisher_average_pages("Penguin")
print(f"{stats['publisher']}: Avg {stats['average_pages']} pages")

                          RESPONSE CODE REFERENCE
================================================================================

200 - OK (Request successful)

204 - No Content (Deletion successful)
400 - Bad Request (Invalid parameters)
404 - Not Found (Resource doesn't exist)
409 - Conflict (Duplicate ID or constraint violation)
422 - Unprocessable Entity (Validation error)
500 - Internal Server Error (Server error)
